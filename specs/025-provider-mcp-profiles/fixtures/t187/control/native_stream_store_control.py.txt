def sha(path: pathlib.Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def pin(path: pathlib.Path, expected: str) -> None:
    st = path.lstat()
    if stat.S_ISLNK(st.st_mode) or not stat.S_ISREG(st.st_mode) or st.st_nlink != 1 or sha(path) != expected:
        raise RuntimeError("pinned_input_mismatch")

def private_write(path: pathlib.Path, raw: bytes) -> None:
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    os.fchmod(fd, 0o600)
    with os.fdopen(fd, "wb") as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())

def row_summary(row: dict) -> dict:
    rid = hashlib.sha256(f"{row['id']}\0{row['seq']}".encode()).hexdigest()
    return {key: row[key] for key in ("seq", "data_bytes", "time_type", "created_type", "completed_type", "completed_positive", "finish_type", "finish_class", "content_type", "part_types", "text_bytes")} | {"row_key_sha256": rid}


def retire_group(proc: subprocess.Popen, wait_seconds: float = 1.0) -> bool:
    if proc.poll() is None:
        try: os.killpg(proc.pid, signal.SIGTERM)
        except ProcessLookupError: pass
        try: proc.wait(timeout=wait_seconds)
        except subprocess.TimeoutExpired:
            try: os.killpg(proc.pid, signal.SIGKILL)
            except ProcessLookupError: pass
            try: proc.wait(timeout=wait_seconds)
            except subprocess.TimeoutExpired: pass
    try: os.killpg(proc.pid, 0)
    except ProcessLookupError: return True
    try: os.killpg(proc.pid, signal.SIGTERM)
    except ProcessLookupError: return True
    end = time.monotonic() + wait_seconds
    while time.monotonic() < end:
        try: os.killpg(proc.pid, 0)
        except ProcessLookupError: return True
        time.sleep(0.05)
    try: os.killpg(proc.pid, signal.SIGKILL)
    except ProcessLookupError: return True
    try: proc.wait(timeout=wait_seconds)
    except subprocess.TimeoutExpired: pass
    try: os.killpg(proc.pid, 0)
    except ProcessLookupError: return True
    return False

    emit({"schema": "caos.spec025.t168.synthetic-stream-store-observation.v1", "source_head": HEAD, "binary_sha256": BINARY_SHA, "native_elapsed_seconds": round(time.monotonic()-start, 3), "native_timed_out": timed_out, "native_exit_code": exit_code, "process_group_retired": retired, "provider": provider, "holds": {name: aggregate(items) for name, items in samples.items()}, "same_incomplete_row_growth": growth, "after_finish": {"reader_state": final["state"], "assistant_row_count": len(final["rows"]), "assistant_rows": [row_summary(row) for row in final["rows"]]}, "completed_row_same_as_partial": complete_same, "strict_completed_reader_ok": reader_ok, "strict_completed_parts": part_count, "strict_completed_text_bytes": total_bytes})

def aggregate(samples: list[dict]) -> dict:
    fields = ("reader_state", "time_type", "created_type", "completed_type", "finish_type", "finish_class", "content_type")
    counts = {field: collections.Counter() for field in fields}
    row_counts, observations, prior = collections.Counter(), [], None
    for sample in samples:
        counts["reader_state"][sample["reader_state"]] += 1
        row_counts[str(len(sample["assistant_rows"]))] += 1
        sig = json.dumps(sample["assistant_rows"], sort_keys=True, separators=(",", ":"))
        if sig != prior:
            observations.append({"elapsed_ms": sample["elapsed_ms"], "assistant_rows": sample["assistant_rows"]}); prior = sig
        for row in sample["assistant_rows"]:
            for field in fields[1:]: counts[field][row[field]] += 1
    return {"samples": len(samples), "assistant_row_counts": dict(row_counts), "categories": {k: dict(v) for k, v in counts.items()}, "changed_observations": observations}

def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("--repo-root", type=pathlib.Path); p.add_argument("--binary", type=pathlib.Path)
    p.add_argument("--execute-synthetic-native", action="store_true"); p.add_argument("--script-sha")
    p.add_argument("--inside", action="store_true"); p.add_argument("--run-root", type=pathlib.Path)
    p.add_argument("--nonce"); p.add_argument("--diag"); p.add_argument("--token")
    a = p.parse_args()
    if a.inside:
        if not all((a.run_root, a.nonce, a.diag, a.token, a.script_sha)): raise RuntimeError("inner_arguments_incomplete")
        return inner(a)
    if a.repo_root is None or a.binary is None: p.error("--repo-root and --binary required")
    return host(a)
