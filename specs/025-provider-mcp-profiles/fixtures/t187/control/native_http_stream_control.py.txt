def sha(path: pathlib.Path) -> str:
    info = path.lstat()
    if not stat.S_ISREG(info.st_mode) or stat.S_ISLNK(info.st_mode) or info.st_nlink != 1:
        raise RuntimeError("pinned_file_identity_invalid")
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def pin(path: pathlib.Path, expected: str) -> None:
    if sha(path) != expected:
        raise RuntimeError("pinned_input_mismatch")


def private_write(path: pathlib.Path, raw: bytes, mode: int = 0o600) -> None:
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, mode)
    os.fchmod(fd, mode)
    with os.fdopen(fd, "wb") as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())


def replace_private_constants(path: pathlib.Path) -> str:
    info = path.lstat()
    if (
        not stat.S_ISREG(info.st_mode)
        or stat.S_ISLNK(info.st_mode)
        or info.st_nlink != 1
        or stat.S_IMODE(info.st_mode) != 0o600
    ):
        raise RuntimeError("private_module_identity_invalid")
    raw = path.read_bytes()
    replacements = (
        (b'const PROVIDER_ID = "opencode-go"', b'const PROVIDER_ID = "t168fake"'),
        (b'const MODEL_ID = "glm-5.3-flash"', b'const MODEL_ID = "stream-probe"'),
    )
    for old, new in replacements:
        if raw.count(old) != 1:
            raise RuntimeError("private_provider_specialization_marker_invalid")
        raw = raw.replace(old, new, 1)
    fd = os.open(path, os.O_WRONLY | os.O_TRUNC | os.O_NOFOLLOW)
    try:
        os.fchmod(fd, 0o600)
        with os.fdopen(fd, "wb", closefd=False) as stream:
            stream.write(raw)
            stream.flush()
            os.fsync(stream.fileno())
    finally:
        os.close(fd)
    return sha(path)

def formatted_prompt(BaseProvider, actual_message: str, token: str) -> tuple[str, str, str]:
    body_marker, receipt_marker = "__SOURCE_BODY__", "__SOURCE_RECEIPT__"
    template = BaseProvider._format_turn_receipt_input(
        SimpleNamespace(terminal_id="{{CAO_TERMINAL_ID}}"), body_marker, receipt_marker
    )
    if template.count(body_marker) != 1 or template.count(receipt_marker) != 1:
        raise RuntimeError("receipt_formatter_markers_invalid")
    suffix_from_body = template.split(body_marker, 1)[1]
    receipt_prefix, receipt_suffix = suffix_from_body.split(receipt_marker, 1)
    prepared = BaseProvider._format_turn_receipt_input(
        SimpleNamespace(terminal_id=TERMINAL_ID), actual_message, token
    )
    if prepared.count(token) != 1 or "{{CAO_TERMINAL_ID}}" in prepared:
        raise RuntimeError("native_prompt_receipt_invalid")
    return prepared, receipt_prefix, receipt_suffix

def retire_group(proc: subprocess.Popen, wait_seconds: float = 1.0) -> bool:
    if proc.poll() is None:
        try:
            os.killpg(proc.pid, signal.SIGTERM)
        except ProcessLookupError:
            pass
        try:
            proc.wait(timeout=wait_seconds)
        except subprocess.TimeoutExpired:
            try:
                os.killpg(proc.pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
            try:
                proc.wait(timeout=wait_seconds)
            except subprocess.TimeoutExpired:
                pass
    try:
        os.killpg(proc.pid, 0)
    except ProcessLookupError:
        return True
    try:
        os.killpg(proc.pid, signal.SIGTERM)
    except ProcessLookupError:
        return True
    deadline = time.monotonic() + wait_seconds
    while time.monotonic() < deadline:
        try:
            os.killpg(proc.pid, 0)
        except ProcessLookupError:
            return True
        time.sleep(0.05)
    try:
        os.killpg(proc.pid, signal.SIGKILL)
    except ProcessLookupError:
        return True
    try:
        proc.wait(timeout=wait_seconds)
    except subprocess.TimeoutExpired:
        pass
    try:
        os.killpg(proc.pid, 0)
    except ProcessLookupError:
        return True
    return False

                except (BrokenPipeError, ConnectionResetError, OSError):

                        state["error"] = "native_closed_stream"

                        if state["stage"] != "finish_sent":
                            state["stage"] = "request_error"

        def do_GET(self):
            self.reply_error(404)

def host(args) -> int:
    script = pathlib.Path(__file__)
    info = script.lstat()
    if (
        script.parent != DRAFT
        or stat.S_ISLNK(info.st_mode)
        or stat.S_IMODE(info.st_mode) != 0o600
        or info.st_nlink != 1
    ):
        raise RuntimeError("private_control_file_metadata_mismatch")
    script_digest = sha(script)
    pins = check_inputs(args.repo_root, args.binary)
    if not args.execute_synthetic_native:
        emit(
            {
                "state": "closed",
                "native_started": False,
                "script_sha256": script_digest,
                "source_head": HEAD,
                "pins": pins,
            }
        )
        return 0
    if not args.script_sha or args.script_sha != script_digest:
        raise RuntimeError("explicit_matching_script_sha_required")

    def available_memory_mib() -> int:
        with pathlib.Path("/proc/meminfo").open() as stream:
            return (
                int(next(line.split()[1] for line in stream if line.startswith("MemAvailable:")))
                // 1024
            )

    available = available_memory_mib()
    if available < MEMORY_FLOOR_MIB:
        raise RuntimeError("memory_floor_not_met")
    run_id = uuid.uuid4().hex
    runs = DRAFT / "runs"
    if runs.exists():
        info = runs.lstat()
        if (
            not stat.S_ISDIR(info.st_mode)
            or stat.S_ISLNK(info.st_mode)
            or info.st_uid != os.getuid()
            or stat.S_IMODE(info.st_mode) != 0o700
        ):
            raise RuntimeError("private_runs_directory_invalid")
    else:
        runs.mkdir(mode=0o700)
    run = runs / run_id
    run.mkdir(mode=0o700)
    for name in ("data", "cache", "state", "tmp", "home", "xdg-config"):
        (run / name).mkdir(mode=0o700)
    (run / "data/opencode").mkdir(mode=0o700)
    private_write(run / "hosts", b"127.0.0.1 localhost\n")
    private_write(
        run / "passwd",
        b"root:x:0:0:root:/home/t168:/bin/sh\nt168:x:1000:1000:t168:/home/t168:/bin/sh\n",
    )
    setup = setup_run(args.repo_root, run, run_id)
    runtime, config = setup["runtime"], setup["config_path"]
    # The ordinary project directory is writable just as in the real worker.
    # Freeze code/expectation inputs independently of native generated metadata.
    input_paths = [
        setup["loader_path"],
        config,
        run / "http-prompt.txt",
        *runtime.rglob("*.js"),
        runtime / ".opencode/http-prompt-expectations.json",
        *(run / "python").rglob("*.py"),
    ]
    input_hashes = {str(path.relative_to(run)): sha(path) for path in input_paths}
    # The path created by installer is mode 0700. The live ledger remains writable.
    for relative in ("data", "cache", "state", "tmp", "home", "xdg-config"):
        if not (run / relative).is_dir():
            raise RuntimeError("runtime_directory_missing")
    args_bwrap = prepare_bwrap(
        run,
        runtime,
        config,
        script,
        script_digest,
        setup["nonce"],
        setup["diagnostic"],
        setup["receipt_sha256"],
    )
    if available_memory_mib() < MEMORY_FLOOR_MIB:
        raise RuntimeError("memory_floor_not_met_at_launch")
    proc = subprocess.Popen(
        args_bwrap,
        cwd=str(args.repo_root),
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.DEVNULL,
        env={"PATH": "/usr/bin:/bin"},
        start_new_session=True,
    )
    outer_timeout = False
    try:
        output, _ = proc.communicate(timeout=OUTER_DEADLINE_SECONDS)
    except subprocess.TimeoutExpired:
        outer_timeout = True
        output = b""
    outer_group_retired = retire_group(proc)
    if outer_timeout:
        try:
            output, _ = proc.communicate(timeout=1)
        except subprocess.TimeoutExpired:
            output = b""
    try:
        observation = json.loads(output.decode("utf-8"))
    except (UnicodeError, ValueError):
        observation = {"state": "inner_receipt_unavailable"}
    try:
        input_preserved = all(sha(run / name) == digest for name, digest in input_hashes.items())
        source_preserved = sha(script) == script_digest and all(
            sha(args.repo_root / name) == digest for name, digest in PINS.items()
        )
    except (OSError, RuntimeError):
        input_preserved = source_preserved = False
    capability_pass = bool(
        isinstance(observation, dict)
        and observation.get("native_exit_code") == 0
        and observation.get("native_timed_out") is False
        and observation.get("process_group_retired") is True
        and observation.get("text_growth_observed") is True
        and observation.get("two_distinct_positive_sequences") is True
        and observation.get("same_observer_request_directory_bindings") == 1
        and observation.get("provider", {}).get("error") is None
        and observation.get("after_finish_projection_state")
        in {"stream_finished", "observer_closed"}
        and not outer_timeout
        and outer_group_retired
        and input_preserved
        and source_preserved
    )
    receipt = {
        "schema": "caos.spec025.t168.private-http-control-run.v1",
        "run_id": run_id,
        "memory_admission_mib": available,
        "pins": pins,
        "input_hashes_before": input_hashes,
        "private_inputs_preserved": input_preserved,
        "source_preserved": source_preserved,
        "native_clone_capability_pass": capability_pass,
        "t168_accepted": False,
        "derivation": setup["derivation"],
        "bwrap_exit_code": proc.returncode,
        "outer_timeout": outer_timeout,
        "outer_process_group_retired": outer_group_retired,
        "observation": observation,
    }
    private_write(
        run / "receipt.json",
        json.dumps(receipt, sort_keys=True, separators=(",", ":")).encode() + b"\n",
    )
    emit(
        {
            "state": "finished",
            "run_id": run_id,
            "bwrap_exit_code": proc.returncode,
            "receipt": str(run / "receipt.json"),
        }
    )
    return 0 if proc.returncode == 0 else 2

    parser.add_argument("--nonce")
    parser.add_argument("--diagnostic")
    parser.add_argument("--receipt-sha256")

                args.run_root,
                args.run_path,

                args.nonce,
                args.diagnostic,
                args.receipt_sha256,
                args.script_sha,
            )
        ):
            raise RuntimeError("inner_arguments_incomplete")
        return inner(args)
    return host(args)
