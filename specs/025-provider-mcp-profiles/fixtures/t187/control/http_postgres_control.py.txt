class GateError(RuntimeError):
    pass


def source_gate():
    if Path.cwd() != REPO:
        raise GateError("canonical_workdir_mismatch")
    top = subprocess.run(
        ["git", "rev-parse", "--show-toplevel"],
        cwd=REPO,
        text=True,
        capture_output=True,
        timeout=15,
    )
    head = subprocess.run(
        ["git", "rev-parse", "HEAD"], cwd=REPO, text=True, capture_output=True, timeout=15
    )
    if (
        top.returncode
        or top.stdout.strip() != str(REPO)
        or head.returncode
        or head.stdout.strip() != HEAD
    ):
        raise GateError("repository_head_or_root_mismatch")
    for path, expected in PINS.items():
        if hashlib.sha256((REPO / path).read_bytes()).hexdigest() != expected:
            raise GateError("source_pin_mismatch")


def admit():
    value = next(
        (
            int(x.split()[1]) // 1024
            for x in Path("/proc/meminfo").read_text().splitlines()
            if x.startswith("MemAvailable:")
        ),
        None,
    )
    if value is None or value < 6144:
        raise GateError("memory_below_6144_mib_floor")
    return value


def docker_json(binary, *args):
    result = subprocess.run([binary, *args], text=True, capture_output=True, timeout=15)
    if result.returncode:
        raise GateError("owned_postgres_identity_unavailable")
    return json.loads(result.stdout)


def owned_network(fixtures, binary, service, receipt):
    cid, run = service.metadata["container_id"], service.metadata["run_id"]
    name = f"cao-spec021-test-net-{run}"
    receipt.update(container_id=cid, run_id=run, network_name=name)
    containers = docker_json(binary, "inspect", cid)
    networks = docker_json(binary, "network", "inspect", name)
    if len(networks) == 1:
        receipt["network_id"] = networks[0].get("Id")
    if len(containers) != 1 or len(networks) != 1:
        raise GateError("owned_identity_ambiguous")
    c, n = containers[0], networks[0]
    labels, nlabels = c.get("Config", {}).get("Labels") or {}, n.get("Labels") or {}
    attached = c.get("NetworkSettings", {}).get("Networks") or {}
    if (
        c.get("Id") != cid
        or c.get("Image") != IMAGE_ID
        or c.get("Config", {}).get("Image") != IMAGE
        or c.get("Name") != f"/cao-spec021-pg-{run}"
        or labels.get(fixtures.OWNER_LABEL) != fixtures.OWNER_VALUE
        or labels.get(fixtures.RUN_LABEL) != run
        or list(attached) != [name]
        or n.get("Name") != name
        or n.get("Id") != attached[name].get("NetworkID")
        or nlabels.get(fixtures.OWNER_LABEL) != fixtures.OWNER_VALUE
        or nlabels.get(fixtures.RUN_LABEL) != run
    ):
        raise GateError("owned_identity_mismatch")
    receipt["identity_verified"] = True
    return receipt


def absent(binary, args):
    # A successful empty exact-ID inventory proves absence without relying on
    # Docker CLI's version-dependent localized error messages.
    kind = "network" if args[:2] == ["network", "inspect"] else "container"
    identity = args[-1]
    if len(identity) != 64 or any(c not in "0123456789abcdef" for c in identity):
        return "unknown"
    try:
        result = subprocess.run(
            [binary, kind, "ls", "--no-trunc", "--quiet", "--filter", "id=" + identity]
            + (["--all"] if kind == "container" else []),
            text=True,
            capture_output=True,
            timeout=15,
        )
        if result.returncode != 0:
            return "unknown"
        rows = result.stdout.splitlines()
        if not rows:
            return True
        return False if rows == [identity] else "unknown"
    except (OSError, subprocess.SubprocessError):
        return "unknown"


def mkdir_private(path):
    path.mkdir(mode=0o700)
    os.chmod(path, 0o700)


def write_ledger(root, terminal, nonce, digest):
    current = root
    mkdir_private(current)
    for part in (".opencode", "http-text-stream-observation", terminal):
        current = current / part
        mkdir_private(current)
    sample = {
        "schema_version": 1,
        "nonce": nonce,
        "sequence": 1,
        "request_ordinal": 1,
        "request_bound": True,
        "phase_ordinal": 2,
        "receipt_sha256": digest,
        "state": "active",
        "reason": "none",
        "wire_bytes": 300,
        "text_bytes": 12,
        "text_events": 2,
        "event_count": 3,
        "original_not_replaced": True,
    }
    path = current / "sample-0001.json"
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w", encoding="utf-8") as stream:
        json.dump(sample, stream, separators=(",", ":"))
        stream.flush()
        os.fsync(stream.fileno())
    os.chmod(path, 0o600)


def api_projection(terminal, generation):
    # Synthetic GET-shaped fields; the CAO API is not called.
    return {
        "id": terminal,

        "status": "processing",
        "turn": {
            "terminal_id": terminal,
            "provider": "opencode_cli",
            "state": "pending",

        },
    }


def put_receipt(connect, sql, target, terminal, generation, digest, phase, insert=False):
    table = sql.Identifier(target.schema, "terminal_turn_receipts")
    with connect(**target.connection_parameters()) as db:
        if insert:
            result = db.execute(
                sql.SQL(
                    "INSERT INTO {} (terminal_id,provider,generation,receipt_sha256,phase,created_at,updated_at) VALUES (%s,'opencode_cli',%s,%s,%s,now(),now())"
                ).format(table),
                (terminal, generation, digest, phase),
            )
        else:
            result = db.execute(
                sql.SQL(
                    "UPDATE {} SET generation=%s,receipt_sha256=%s,phase=%s,updated_at=now() WHERE terminal_id=%s"
                ).format(table),
                (generation, digest, phase, terminal),
            )
        if result.rowcount != 1:
            raise GateError("receipt_write_count_mismatch")


def capture(activity, psycopg, server, root, terminal, metadata, nonce):
    connect, readonly = psycopg.connect, []

    def observed_connect(**params):
        conn = connect(**params)
        if params.get("application_name") == "t187-http-stream-readonly":
            state = conn.execute("SHOW transaction_read_only").fetchone()[0]
            readonly.append(state)
            if state != "on":
                conn.close()
                raise GateError("capture_connection_not_read_only")
        return conn

    psycopg.connect = observed_connect
    try:
        result = activity.capture(server, root, terminal, metadata, nonce, time.monotonic() + 10)
    finally:
        psycopg.connect = connect
    return result, readonly


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    if Path(module.__file__) != path:
        raise GateError("import_root_mismatch")
    return module


def verify_race(raced, reads, wrote, readonly, persisted, replacement):
    if (
        raced is not None
        or reads != 1
        or wrote is not True
        or readonly != ["on", "on"]
        or persisted != [(replacement,)]
    ):
        raise GateError("generation_race_or_readonly_check_failed")

def stop(_signum, _frame):
    signal.signal(signal.SIGTERM, signal.SIG_IGN)
    raise KeyboardInterrupt


def main():
    if (
        len(sys.argv) != 4
        or sys.argv[1:3] != ["--execute", "--script-sha"]
        or sys.argv[3] != hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    ):
        print("gate_closed: explicit --execute required")
        return 2
    signal.signal(signal.SIGTERM, stop)
    try:
        result = run()
    except KeyboardInterrupt:
        print("diagnostic_interrupted", file=sys.stderr)
        return 130
    except GateError as error:
        print("diagnostic_failed:" + str(error), file=sys.stderr)
        return 1
    except Exception as error:
        print("diagnostic_failed:" + type(error).__name__, file=sys.stderr)
        return 1
    print(
        json.dumps(
            {
                "status": "diagnostic_only",
                "source_head": HEAD,
                "api_metadata": "synthetic GET-shaped projection; API not called",
                "native_provider_invoked": False,
                "model_invoked": False,
                "sqlite_used": False,
                "migration_path": "fixture bootstrap_scope -> core_migrations -> migrate_component",
                **result,
            },
            sort_keys=True,
        )
    )
    return 0


if __name__ == "__main__":
