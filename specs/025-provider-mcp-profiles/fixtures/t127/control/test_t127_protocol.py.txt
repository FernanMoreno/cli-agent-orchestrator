    def test_expectations_are_bound_to_authoritative_public_phase_source(self) -> None:
        binding = json.loads(
            (CONTROL_DIR / "source-binding.json").read_text(encoding="utf-8")
        )
        repo_root = Path(os.environ["T127_TEST_REPO_ROOT"]).resolve(strict=True)
        self.assertEqual(binding["source_head"], "6ffaefddf26cc0bb0c003dd487bfaba4e04e1a69")
        raw = (repo_root / binding["source_file_relative"]).read_bytes()
        self.assertTrue(
            hashlib.sha256(raw).hexdigest() == binding["source_file_sha256"]
        )
        git_blob = hashlib.sha1(
            b"blob " + str(len(raw)).encode("ascii") + b"\0" + raw
        ).hexdigest()
        self.assertTrue(git_blob == binding["source_git_blob_sha1"])
        module = ast.parse(raw.decode("utf-8", "strict"))
        node = next(
            statement
            for statement in module.body
            if isinstance(statement, ast.Assign)
            and any(
                isinstance(target, ast.Name) and target.id == "PRE_RESTART_MESSAGE"
                for target in statement.targets
            )
        )
        source_message = ast.literal_eval(node.value)
        baseline = self.expectations["source_baseline"]
        mutant = self.expectations["unicode_punctuation_mutant"]
        self.assertTrue(baseline["source_message"] == source_message)
        self.assertTrue(baseline["wire_message"] == source_message)
        self.assertTrue(mutant["source_message"] == source_message)
        self.assertTrue(mutant["wire_message"] == source_message[:-1] + "。")
        self.assertEqual(len(source_message), 40)
        self.assertEqual(len(source_message.encode("utf-8")), 48)
        self.assertEqual(len(mutant["wire_message"].encode("utf-8")), 50)
        import driver

        driver._verify_source_binding(binding["source_head"], repo_root)
        with self.assertRaises(RuntimeError):
            driver._verify_source_binding("0" * 40, repo_root)

    def _write_probe_ledger(self, root: Path, workspace: Path, *, duplicate: bool = False) -> None:
        opencode = workspace / ".opencode"
        plugin_dir = opencode / "plugins"
        opencode.mkdir(mode=0o700)
        plugin_dir.mkdir(mode=0o700)
        plugin_source = (
            CONTROL_DIR / "workspaces" / "source_baseline" / ".opencode" / "plugins" / "response_hook_probe.js"
        )
        plugin = plugin_dir / "response_hook_probe.js"
        plugin.write_bytes(plugin_source.read_bytes())
        plugin.chmod(0o600)
        ledger = opencode / "http-response-probe"
        ledger.mkdir(mode=0o700)
        hook = {
            "schema_version": 1,
            "event": "hook",
            "status": "registered",
            "ordinal": 0,
            "hook_registered": True,
            "selected_primary_response_seen": False,
            "synthetic_wire_marker_seen": False,
            "clone_completed": False,
            "inspection_unavailable": False,
            "original_not_replaced": True,
            "cleanup_complete": False,
        }
        responses = [
            {
                "schema_version": 1,
                "event": "response",
                "status": "clone_completed",
                "ordinal": ordinal,
                "hook_registered": True,
                "selected_primary_response_seen": True,
                "synthetic_wire_marker_seen": ordinal == 1,
                "clone_completed": True,
                "inspection_unavailable": False,
                "original_not_replaced": True,
                "cleanup_complete": False,
            }
            for ordinal in (1, 2)
        ]
        if duplicate:
            events = b'{"schema_version":1,"schema_version":1}\n'
        else:
            events = ("\n".join(json.dumps(item, sort_keys=True) for item in [hook, *responses]) + "\n").encode()
        events_path = ledger / "events.jsonl"
        events_path.write_bytes(events)
        events_path.chmod(0o600)
        seal = {
            "schema_version": 1,
            "event": "seal",
            "status": "sealed",
            "ordinal": 2,
            "hook_registered": True,
            "selected_primary_response_seen": True,
            "synthetic_wire_marker_seen": True,
            "clone_completed": True,
            "inspection_unavailable": False,
            "original_not_replaced": True,
            "cleanup_complete": True,
        }
        seal_path = ledger / "seal.json"
        seal_path.write_text(json.dumps(seal, sort_keys=True) + "\n", encoding="utf-8")
        seal_path.chmod(0o600)

    def test_u3002_mutant_negative_compares_against_unmutated_source(self) -> None:
        baseline = self.expectations["source_baseline"]
        mutant = self.expectations["unicode_punctuation_mutant"]
        source_message = baseline["source_message"]
        mutant_message = mutant["wire_message"]
        self.assertEqual(len(source_message), 40)
        self.assertEqual(len(source_message.encode("utf-8")), 48)
        self.assertEqual(len(mutant_message), 40)

        self.assertTrue(mutant_message == source_message[:-1] + "。")

            {"message": source_message, "target": mutant["target"]}, mutant
        )
        self.assertFalse(summary["argument_exact"])
        self.assertFalse(summary["utf8_byte_length_equal"])
        self.assertTrue(summary["unicode_scalar_count_equal"])
        self.assertEqual(summary["failure_class"], "mismatch")
        same_length_change = source_message[:-1] + "!"
        equal_counts = control.summarize_tool_arguments(
            {"message": same_length_change, "target": baseline["target"]}, baseline
        )
        self.assertFalse(equal_counts["argument_exact"])
        self.assertTrue(equal_counts["utf8_byte_length_equal"])
        self.assertTrue(equal_counts["unicode_scalar_count_equal"])
        self.assertNotIn("message", summary)
        self.assertNotIn("hash", summary)

    def test_outer_exception_after_launch_reports_launch_and_reaps_owned_group(self) -> None:
        import driver
        from types import SimpleNamespace

        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary)
            (root / "private").mkdir(mode=0o700)
            class Process:
                pid = 43218
                returncode = None

                def poll(self) -> None:
                    return None

            process = Process()
            killpg_calls: list[tuple[int, int]] = []

            def killpg(process_group: int, signal_number: int) -> None:
                killpg_calls.append((process_group, signal_number))
                raise ProcessLookupError

            output = StringIO()
            with (
                patch.object(driver, "RUN_ENABLED", True),
                patch.object(driver, "PRIVATE_ADMISSION_APPROVED", True),
                patch.object(driver, "CONTROL", root),
                patch.object(driver, "_validate_inputs"),
                patch.object(driver, "_create_runtime_dirs"),
                patch.object(driver, "_make_argv", return_value=["sandbox"]),
                patch.object(
                    driver,
                    "_mem_available_kib",
                    side_effect=[driver.FLOOR_KIB, driver.FLOOR_KIB, RuntimeError("post launch")],
                ),
                patch.object(driver.subprocess, "Popen", return_value=process),
                patch.object(
                    driver,
                    "_cleanup_owned_process_group",
                    wraps=driver._cleanup_owned_process_group,
                ) as cleanup,
                patch.object(driver.os, "killpg", side_effect=killpg),
                patch.object(
                    driver.sys,
                    "argv",
                    ["driver.py", "--execute", "--expected-head", "0" * 40, "--repo-root", "/tmp"],
                ),
                redirect_stdout(output),
            ):
                self.assertEqual(driver.main(), 1)

            projection = json.loads(output.getvalue())
            self.assertTrue(projection["native_invoked"])
            cleanup.assert_called_once_with(process)
            self.assertEqual({group for group, _signal in killpg_calls}, {process.pid})

    def test_outer_cleanup_targets_only_the_owned_process_group(self) -> None:
        import driver
        from types import SimpleNamespace

        calls: list[tuple[int, int]] = []
        alive = True

        def killpg(process_group: int, signal_number: int) -> None:
            nonlocal alive
            calls.append((process_group, signal_number))
            if signal_number == 0:
                if alive:
                    return
                raise ProcessLookupError
            if signal_number == driver.signal.SIGTERM:
                alive = False
                return
            self.fail("cleanup unexpectedly escalated to SIGKILL")

        with patch.object(driver.os, "killpg", side_effect=killpg):
            result = driver._cleanup_owned_process_group(SimpleNamespace(pid=43217))
        self.assertEqual(result, {
            "owned_process_group_cleanup_attempted": True,
            "owned_process_group_remaining": False,
        })
        self.assertTrue(calls)
        self.assertEqual({process_group for process_group, _signal in calls}, {43217})
        self.assertNotIn((43217, driver.signal.SIGKILL), calls)
        self.assertFalse(hasattr(driver, "_scan_and_reap_marker"))

    def test_mcp_summary_is_available_after_signal_without_stdin_eof(self) -> None:
        expected = self.expectations["source_baseline"]
        request = {
            "jsonrpc": "2.0",
            "id": 2,
            "method": "tools/call",
            "params": {
                "name": "send_message",
                "arguments": {
                    "message": expected["wire_message"],
                    "target": expected["target"],
                },
            },
        }
        rejected_extra = {
            "jsonrpc": "2.0",
            "id": 3,
            "method": "tools/call",
            "params": {"name": "other_tool", "arguments": {}},
        }

        class SignalAfterCalls:
            def __init__(self, lines: list[bytes], summary_path: Path) -> None:
                self.lines = iter(lines)
                self.handler = None
                self.summary_path = summary_path
                self.initial_state_seen = False

            def readline(self, _limit: int) -> bytes:
                try:
                    if not self.initial_state_seen:
                        initial = control._read_mcp_summary(
                            self.summary_path, "source_baseline", 1
                        )
                        if (
                            initial is None
                            or initial["tool_call_attempt_count"] != 0
                            or initial["fake_mcp_dispatch_count"] != 0
                        ):
                            raise AssertionError("initial MCP summary was not available")
                        self.initial_state_seen = True
                    return next(self.lines)
                except StopIteration:
                    if self.handler is None:
                        raise AssertionError("SIGTERM handler was not installed")
                    self.handler(control.signal.SIGTERM, None)
                    raise AssertionError("SIGTERM handler returned")

        class Input:
            def __init__(self, buffer: SignalAfterCalls) -> None:
                self.buffer = buffer

        class OutputBuffer:
            def write(self, _data: bytes) -> int:
                return len(_data)

            def flush(self) -> None:
                return None

        class Output:
            buffer = OutputBuffer()

        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary)
            root.chmod(0o700)
            private = root / "private"
            private.mkdir(mode=0o700)
            expectation_path = root / "expectations.json"
            expectation_path.write_text(
                (CONTROL_DIR / "expectations.json").read_text(encoding="utf-8"),
                encoding="utf-8",
            )
            expectation_path.chmod(0o600)
            summary_path = private / "mcp-summary-source_baseline.json"
            stream = SignalAfterCalls(
                [
                    json.dumps(request, separators=(",", ":")).encode() + b"\n",
                    json.dumps(rejected_extra, separators=(",", ":")).encode() + b"\n",
                ],
                summary_path,
            )
            real_signal = control.signal.getsignal

            def install(sig: int, handler: object) -> object:
                previous = real_signal(sig)
                stream.handler = handler
                return previous

            with (
                patch.object(control, "ROOT", root),
                patch.object(control, "EXPECTATIONS", expectation_path),
                patch.object(control.sys, "stdin", Input(stream)),
                patch.object(control.sys, "stdout", Output()),
                patch.object(control.signal, "signal", side_effect=install),
                patch.dict(
                    os.environ,
                    {
                        "T127_CASE": "source_baseline",
                        "T127_EVIDENCE_PATH": str(private / "callback-source_baseline.jsonl"),
                    },
                    clear=False,
                ),
            ):
                self.assertEqual(control.run_mcp_stdio(), 2)
            summary = control._read_mcp_summary(
                private / "mcp-summary-source_baseline.json", "source_baseline", 1
            )
            self.assertEqual(summary["tool_call_attempt_count"], 2)
            self.assertEqual(summary["fake_mcp_dispatch_count"], 1)
            self.assertTrue(stream.initial_state_seen)
            self.assertEqual(summary_path.stat().st_mode & 0o777, 0o600)
            self.assertEqual(summary_path.stat().st_nlink, 1)
            self.assertEqual(
                set(summary),
                {"case_ordinal", "case_kind", "tool_call_attempt_count", "fake_mcp_dispatch_count"},
            )

    def test_fake_model_context_stays_above_native_auto_compaction_buffer(self) -> None:

            float("inf") if input_limit is None else input_limit - buffer_tokens,

        self.assertEqual(context, 65_536)
        self.assertEqual(output, 256)

                patch.object(control, "ROOT", private_root),

        runtime_limit = runtime_config["providers"]["t127fake"]["models"][
            "toolarg-model"

        self.assertEqual(runtime_limit["output"], 256)

    def test_invalid_offer_counter_is_not_reported_as_zero(self) -> None:

            self.assertIsNone(projected["parsed_request_count"])

            self.assertFalse(rejected["provider_tool_offer_diagnostics_valid"])

        finally:
